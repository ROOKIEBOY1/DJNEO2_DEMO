cmake_minimum_required(VERSION 3.5.0)
project(neovideo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(neovideo SHARED VideoPreview.cpp HevcStream.cpp MovieExport.cpp MovieFrames.cpp MovieAudio.cpp)
# Keep per-frame conversion and audio resampling responsive in the signed debug demo too.
set_source_files_properties(MovieExport.cpp MovieFrames.cpp MovieAudio.cpp PROPERTIES COMPILE_OPTIONS "-O2")
target_include_directories(neovideo PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(neovideo PRIVATE -Wall -Wextra -Werror=return-type)
target_link_libraries(neovideo PUBLIC libace_napi.z.so libnative_media_vdec.so libnative_media_venc.so libnative_media_acodec.so libnative_media_avsource.so libnative_media_avdemuxer.so libnative_media_avmuxer.so libnative_media_core.so libnative_media_codecbase.so libnative_window.so libnative_buffer.so libhilog_ndk.z.so)
